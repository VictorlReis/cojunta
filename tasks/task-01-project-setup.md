## Implementation Details

### 1. Scaffold Vite project
Run `npm create vite@latest . -- --template react-ts` (in the current directory since the repo already exists).

### 2. Install dependencies
```bash
npm install react-router-dom @tanstack/react-query @supabase/supabase-js zustand recharts
npm install -D tailwindcss @tailwindcss/vite
```

### 3. Tailwind CSS setup
In `src/index.css`:
```css
@import "tailwindcss";
```

In `vite.config.ts` add the tailwind plugin:
```ts
import tailwindcss from "@tailwindcss/vite";
export default defineConfig({
  plugins: [react(), tailwindcss()],
  // ...
});
```

### 4. shadcn/ui init
Run `npx shadcn@latest init` and configure:
- Style: New York
- Base color: Neutral or Zinc
- CSS variables: Yes
- Path alias: `@/`
- Components path: `src/components/ui`

Then install required components:
```bash
npx shadcn@latest add button input label card dialog select dropdown-menu table badge separator sheet skeleton tabs sonner
```

### 5. Path aliases
In `tsconfig.json` (or `tsconfig.app.json`):
```json
{
  "compilerOptions": {
    "baseUrl": ".",
    "paths": {
      "@/*": ["./src/*"]
    }
  }
}
```
In `vite.config.ts`:
```ts
resolve: {
  alias: {
    "@": path.resolve(__dirname, "./src"),
  },
}
```

### 6. Environment variables
`.env.example`:
```
VITE_SUPABASE_URL=your-supabase-project-url
VITE_SUPABASE_ANON_KEY=your-supabase-anon-key
```

`.env.local` (gitignored):
```
VITE_SUPABASE_URL=
VITE_SUPABASE_ANON_KEY=
```

Add `.env.local` to `.gitignore`.

### 7. Supabase client (`src/lib/supabase.ts`)
```ts
import { createClient } from '@supabase/supabase-js'
import type { Database } from '@/types/database'

const supabaseUrl = import.meta.env.VITE_SUPABASE_URL
const supabaseAnonKey = import.meta.env.VITE_SUPABASE_ANON_KEY

export const supabase = createClient<Database>(supabaseUrl, supabaseAnonKey)
```

### 8. Utility functions (`src/lib/utils.ts`)
Include the `cn()` utility from shadcn plus:
```ts
export function formatCurrency(value: number): string {
  return new Intl.NumberFormat('pt-BR', {
    style: 'currency',
    currency: 'BRL',
  }).format(value)
}

export function formatDate(date: string | Date): string {
  return new Intl.DateTimeFormat('pt-BR').format(new Date(date))
}
```

### 9. Placeholder App.tsx
Set up React Router with placeholder routes and TanStack Query provider:
```tsx
import { BrowserRouter, Routes, Route } from 'react-router-dom'
import { QueryClient, QueryClientProvider } from '@tanstack/react-query'

const queryClient = new QueryClient()

function App() {
  return (
    <QueryClientProvider client={queryClient}>
      <BrowserRouter>
        <Routes>
          <Route path="/" element={<div>Dashboard (placeholder)</div>} />
          <Route path="/login" element={<div>Login (placeholder)</div>} />
          <Route path="/signup" element={<div>Signup (placeholder)</div>} />
          <Route path="/expenses" element={<div>Expenses (placeholder)</div>} />
          <Route path="/settings" element={<div>Settings (placeholder)</div>} />
        </Routes>
      </BrowserRouter>
    </QueryClientProvider>
  )
}
```

### 10. Zustand store placeholder (`src/stores/authStore.ts`)
```ts
import { create } from 'zustand'
import type { User, Session } from '@supabase/supabase-js'

interface AuthState {
  user: User | null
  session: Session | null
  isLoading: boolean
  setUser: (user: User | null) => void
  setSession: (session: Session | null) => void
  setIsLoading: (isLoading: boolean) => void
}

export const useAuthStore = create<AuthState>((set) => ({
  user: null,
  session: null,
  isLoading: true,
  setUser: (user) => set({ user }),
  setSession: (session) => set({ session }),
  setIsLoading: (isLoading) => set({ isLoading }),
}))
```
